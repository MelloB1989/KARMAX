package payments

import (
	"context"
	"errors"
	"testing"
)

type fakeStore struct{ c *Charge }

func (f fakeStore) Charge(context.Context, string) (*Charge, error) { return f.c, nil }

func TestRefund(t *testing.T) {
	st := fakeStore{&Charge{ID: "ch_1", Amount: 4200, Refunded: 200}}
	cases := []struct {
		name   string
		id     string
		amount int64
		want   int64
		err    error
	}{
		{"valid partial", "ch_1", 1000, 1000, nil},
		{"full remaining", "ch_1", 4000, 4000, nil},
		{"empty id", "", 100, 0, ErrInvalidID},
		{"zero amount", "ch_1", 0, 0, ErrInvalidAmount},
		{"negative amount", "ch_1", -5, 0, ErrInvalidAmount},
		{"over refund", "ch_1", 4001, 0, ErrOverRefund},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			got, err := Refund(context.Background(), st, tc.id, tc.amount)
			if !errors.Is(err, tc.err) || got != tc.want {
				t.Fatalf("got (%d, %v), want (%d, %v)", got, err, tc.want, tc.err)
			}
		})
	}
}
